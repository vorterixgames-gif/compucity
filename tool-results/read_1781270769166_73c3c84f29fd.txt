    98→/**
    99→ * Extract the primary storage capacity in GB from a product name.
   100→ * Handles patterns like: 1TB, 1 TB, 2TB, 256GB, 480GB, 960GB, 1.92TB, etc.
   101→ * Returns null if no recognizable storage capacity found.
   102→ */
   103→function extractCapacityGB(name: string): number | null {
   104→  const n = name.replace(/,/g, '.').replace(/(SSD|HDD|DISCO)(\d)/gi, '$1 $2')
   105→  const tbMatch = n.match(/\b(\d+\.?\d*)\s*TB?\b/i)
   106→  if (tbMatch) {
   107→    const val = parseFloat(tbMatch[1])
   108→    if (val >= 0.1 && val <= 100) return val * 1000
   109→  }
   110→  const gbMatch = n.match(/\b(\d{2,4})\s*GB(?!\s*[\/PS])/i)
   111→  if (gbMatch) {
   112→    const val = parseInt(gbMatch[1])
   113→    if (val >= 32 && val <= 16384) return val
   114→  }
   115→  return null
   116→}

   235→/**
   236→ * Apply manual filters to a product list.
   237→ * Logic: AND between different filter groups, OR within the same group.
   238→ * If no filters active for a group, all products pass that group.
   239→ */
   240→function applyManualFilters(products: BuilderProduct[], filters: Record<string, string[]>, slotKey: string): BuilderProduct[] {
   241→  const slotFilterOptions = SLOT_FILTERS[slotKey]
   242→  if (!slotFilterOptions || slotFilterOptions.length === 0) return products
   243→
   244→  // Get active filter groups
   245→  const activeGroups = new Map<string, FilterOption[]>()
   246→  for (const [key, values] of Object.entries(filters)) {
   247→    if (values.length === 0) continue
   248→    const matchingOptions = slotFilterOptions.filter(o => o.key === key && values.includes(o.value))
   249→    if (matchingOptions.length > 0) {
   250→      activeGroups.set(key, matchingOptions)
   251→    }
   252→  }
   253→
   254→  if (activeGroups.size === 0) return products
   255→
   256→  return products.filter(product => {
   257→    // Product must pass ALL active filter groups (AND logic between groups)
   258→    for (const [, options] of activeGroups) {
   259→      // Product must match at least one option in the group (OR logic within group)
   260→      const matchesGroup = options.some(opt => opt.matchFn(product.name))
   261→      if (!matchesGroup) return false
   262→    }
   263→    return true
   264→  })
   265→}
