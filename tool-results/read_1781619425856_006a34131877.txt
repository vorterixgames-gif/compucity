   217→   217→/**
   218→   218→ * Build a category lookup from the database.
   219→   219→ * Returns slug -> id map, name -> id map, and parent info for subcategory logic.
   220→   220→ */
   221→   221→async function buildCategoryLookup(): Promise<{
   222→   222→  slugToId: Record<string, string>
   223→   223→  nameToId: Record<string, string>
   224→   224→  idToParentId: Record<string, string | null>
   225→   225→  parentSlugToChildSlugs: Record<string, string[]>
   226→   226→}> {
   227→   227→  const result = await db.execute('SELECT id, name, slug, parentId FROM categories')
   228→   228→  const slugToId: Record<string, string> = {}
   229→   229→  const nameToId: Record<string, string> = {}
   230→   230→  const idToParentId: Record<string, string | null> = {}
   231→   231→  const parentSlugToChildSlugs: Record<string, string[]> = {}
   232→   232→  for (const row of result.rows as any[]) {
   233→   233→    if (row.slug) slugToId[row.slug] = row.id
   234→   234→    if (row.name) nameToId[row.name.toLowerCase()] = row.id
   235→   235→    idToParentId[row.id] = row.parentId || null
   236→   236→    if (row.parentId) {
   237→   237→      // Find parent slug
   238→   238→      const parentRow = (result.rows as any[]).find((r: any) => r.id === row.parentId)
   239→   239→      if (parentRow?.slug) {
   240→   240→        if (!parentSlugToChildSlugs[parentRow.slug]) parentSlugToChildSlugs[parentRow.slug] = []
   241→   241→        parentSlugToChildSlugs[parentRow.slug].push(row.slug)
   242→   242→      }
   243→   243→    }
   244→   244→  }
   245→   245→  return { slugToId, nameToId, idToParentId, parentSlugToChildSlugs }
   246→   246→}

   252→   252→async function buildSupplierMappingLookup(supplierId: string): Promise<Record<string, string>> {
   253→   253→  const result = await db.execute({
   254→   254→    sql: 'SELECT supplierCategory, storeCategoryId FROM supplier_category_mappings WHERE supplierId = ?',
   255→   255→    args: [supplierId],
   256→   256→  })
   257→   257→  const map: Record<string, string> = {}
   258→   258→  for (const row of result.rows as any[]) {

   260→   260→  }
   261→   261→  return map
   262→   262→}

   264→   264→/**
   265→   265→ * Map a product to a store category using:
   266→   266→ * 1. Supplier category mapping (if available)
   267→   267→ * 2. Keyword matching with slug-based lookup
   268→   268→ * 3. Subcategory refinement: if the matched category is a parent with subcategories,
   269→   269→ *    use subcategory rules to find the best subcategory match
   270→   270→ * 4. Default: null (no category)
   271→   271→ */
   272→   272→function mapProductToCategory(
   273→   273→  productName: string,
   274→   274→  supplierCategory: string | null,
   275→   275→  supplierMappings: Record<string, string>,
   276→   276→  slugToId: Record<string, string>,
   277→   277→  idToParentId: Record<string, string | null>,
   278→   278→  parentSlugToChildSlugs: Record<string, string[]>,
   279→   279→  slugToParentSlug?: Record<string, string>
   280→   280→): { categoryId: string | null; method: string } {
   281→   281→  let matchedCategoryId: string | null = null
   282→   282→  let method = 'none'
   283→   283→
   284→   284→  // 1. Check supplier category mapping first
   285→   285→  if (supplierCategory && supplierMappings[supplierCategory]) {
   286→   286→    matchedCategoryId = supplierMappings[supplierCategory]
   287→   287→    method = 'mapping'
   288→   288→  }
   289→   289→
   290→   290→  // 2. Keyword matching (only if no mapping match)
   291→   291→  if (!matchedCategoryId) {
   292→   292→    const name = (productName || '').toUpperCase()
   293→   293→    for (const mapping of CATEGORY_KEYWORD_MAP) {
   294→   294→      if (mapping.keywords.some(kw => name.includes(kw))) {
   295→   295→        const categoryId = slugToId[mapping.categorySlug]
   296→   296→        if (categoryId) {
   297→   297→          matchedCategoryId = categoryId
   298→   298→          method = 'keyword'
   299→   299→          break
   300→   300→        }
   301→   301→      }
   302→   302→    }
   303→   303→  }
   304→   304→
   305→   305→  if (!matchedCategoryId) {
   306→   306→    return { categoryId: null, method: 'none' }
   307→   307→  }
   308→   308→
   309→   309→  // 2.5. Category correction: if a product was mapped to a category by supplier mapping
   310→   310→  // but its name indicates it should be in a different category, correct it.
   311→   311→  // This handles cases like "Plataforma AMD/Intel" mapping to microprocesadores

  1214→  1214→/**
  1215→  1215→ * Strip PHP notices/warnings/errors from Air Intra API response text.
  1216→  1216→ * Air Intra's PHP server injects HTML-formatted PHP notices like:
  1217→  1217→ *   <br /> <b>Notice</b>: Undefined property: stdClass::$estado in /path/file.php on line 54
  1218→  1218→ * These can appear BEFORE, AFTER, or INSIDE the JSON array (between product objects).
  1219→  1219→ * This function removes them while preserving the actual JSON data.
  1220→  1220→ */
  1221→  1221→function stripPhpNotices(text: string): string {
  1222→  1222→  let cleaned = text
  1223→  1223→    // Step 1: Remove <b> and </b> tags FIRST.
  1224→  1224→    // Air Intra wraps file paths AND line numbers in <b> tags:
  1225→  1225→    //   "in <b>/path/file.php</b> on line <b>54</b>"
  1226→  1226→    // The old regex expected "on line \d+" (plain digits) and failed to match,
  1227→  1227→    // leaving notices inside JSON objects and making them unparseable.
  1228→  1228→    .replace(/<\/?b>/gi, '')
  1229→  1229→    // Step 2: Remove <br /> and <br> tags
  1230→  1230→    .replace(/<br\s*\/?>\s*/gi, '')
  1231→  1231→    // Step 3: Remove PHP notice/warning/error blocks (now plain text after HTML tag removal).
  1232→  1232→    // Matches: "Notice: <any text> on line <digits>" — non-greedy so it stops at the
  1233→  1233→    // first "on line NNN" (one notice at a time, even if multiple are consecutive).
  1234→  1234→    .replace(/(?:Notice|Warning|Fatal error|Parse error|Deprecated):\s*.*?on line\s+\d+\s*/gis, '')
  1235→  1235→    // Step 4: Fix trailing commas before ] or } (happens when a notice is removed from
  1236→  1236→    // between the last value and the closing brace, e.g. {"k":"v",Notice:...} → {"k":"v",})
  1237→  1237→    .replace(/,\s*([}\]])/g, '$1')
  1238→  1238→    // Step 5: Fix missing commas between objects: }{ should be },{
  1239→  1239→    .replace(/}\s*{/g, '},{')
  1240→  1240→    // Step 6: Fix double commas (happens when a notice is removed from between two commas)
  1241→  1241→    .replace(/,\s*,/g, ',')
  1242→  1242→
  1243→  1243→  return cleaned.trim()
  1244→  1244→}

  1334→  1334→/**
  1335→  1335→ * Parse a potentially corrupted Air Intra JSON array by extracting individual
  1336→  1336→ * product objects using regex. This is the ultimate fallback when the entire

  1365→  1365→        continue

  1368→  1368→        esc = true
  1369→  1369→        continue
  1370→  1370→      }

  1376→  1376→

  1378→  1378→      else if (ch === '}') {
  1379→  1379→        depth--
  1380→  1380→        if (depth === 0) {
  1381→  1381→          objEnd = j
  1382→  1382→          break
  1383→  1383→        }
  1384→  1384→      }
  1385→  1385→    }

  1388→  1388→      i++

  1390→  1390→    }

  1395→  1395→    // (e.g., error objects, impuesto_iva fragments, etc.)
  1396→  1396→    if (objText.includes('"codigo"') || objText.includes('"codiart"')) {
  1397→  1397→      try {
  1398→  1398→        const obj = JSON.parse(objText)
  1399→  1399→        products.push(obj)
  1400→  1400→      } catch {
  1401→  1401→        // Object itself is corrupted, skip it
  1402→  1402→      }
  1403→  1403→    }
  1404→  1404→
  1405→  1405→    i = objEnd + 1
  1406→  1406→  }
  1407→  1407→
  1408→  1408→  return products
  1409→  1409→}
