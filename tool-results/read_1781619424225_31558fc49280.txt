   217→/**
   218→ * Build a category lookup from the database.
   219→ * Returns slug -> id map, name -> id map, and parent info for subcategory logic.
   220→ */
   221→async function buildCategoryLookup(): Promise<{
   222→  slugToId: Record<string, string>
   223→  nameToId: Record<string, string>
   224→  idToParentId: Record<string, string | null>
   225→  parentSlugToChildSlugs: Record<string, string[]>
   226→}> {
   227→  const result = await db.execute('SELECT id, name, slug, parentId FROM categories')
   228→  const slugToId: Record<string, string> = {}
   229→  const nameToId: Record<string, string> = {}
   230→  const idToParentId: Record<string, string | null> = {}
   231→  const parentSlugToChildSlugs: Record<string, string[]> = {}
   232→  for (const row of result.rows as any[]) {
   233→    if (row.slug) slugToId[row.slug] = row.id
   234→    if (row.name) nameToId[row.name.toLowerCase()] = row.id
   235→    idToParentId[row.id] = row.parentId || null
   236→    if (row.parentId) {
   237→      // Find parent slug
   238→      const parentRow = (result.rows as any[]).find((r: any) => r.id === row.parentId)
   239→      if (parentRow?.slug) {
   240→        if (!parentSlugToChildSlugs[parentRow.slug]) parentSlugToChildSlugs[parentRow.slug] = []
   241→        parentSlugToChildSlugs[parentRow.slug].push(row.slug)
   242→      }
   243→    }
   244→  }
   245→  return { slugToId, nameToId, idToParentId, parentSlugToChildSlugs }
   246→}

   248→/**
   249→ * Build supplier category mappings from the database.
   250→ * Returns supplierCategory -> storeCategoryId map for a given supplier.
   251→ */
   252→async function buildSupplierMappingLookup(supplierId: string): Promise<Record<string, string>> {
   253→  const result = await db.execute({
   254→    sql: 'SELECT supplierCategory, storeCategoryId FROM supplier_category_mappings WHERE supplierId = ?',
   255→    args: [supplierId],
   256→  })
   257→  const map: Record<string, string> = {}
   258→  for (const row of result.rows as any[]) {
   259→    map[row.supplierCategory] = row.storeCategoryId
   260→  }
   261→  return map
   262→}

   269→ *    use subcategory rules to find the best subcategory match
   270→ * 4. Default: null (no category)
   271→ */

   274→  supplierCategory: string | null,

  1214→/**
  1215→ * Strip PHP notices/warnings/errors from Air Intra API response text.
  1216→ * Air Intra's PHP server injects HTML-formatted PHP notices like:
  1217→ *   <br /> <b>Notice</b>: Undefined property: stdClass::$estado in /path/file.php on line 54
  1218→ * These can appear BEFORE, AFTER, or INSIDE the JSON array (between product objects).
  1219→ * This function removes them while preserving the actual JSON data.
  1220→ */
  1221→function stripPhpNotices(text: string): string {
  1222→  let cleaned = text
  1223→    // Step 1: Remove <b> and </b> tags FIRST.
  1224→    // Air Intra wraps file paths AND line numbers in <b> tags:
  1225→    //   "in <b>/path/file.php</b> on line <b>54</b>"
  1226→    // The old regex expected "on line \d+" (plain digits) and failed to match,
  1227→    // leaving notices inside JSON objects and making them unparseable.
  1228→    .replace(/<\/?b>/gi, '')
  1229→    // Step 2: Remove <br /> and <br> tags
  1230→    .replace(/<br\s*\/?>\s*/gi, '')
  1231→    // Step 3: Remove PHP notice/warning/error blocks (now plain text after HTML tag removal).
  1232→    // Matches: "Notice: <any text> on line <digits>" — non-greedy so it stops at the
  1233→    // first "on line NNN" (one notice at a time, even if multiple are consecutive).
  1234→    .replace(/(?:Notice|Warning|Fatal error|Parse error|Deprecated):\s*.*?on line\s+\d+\s*/gis, '')
  1235→    // Step 4: Fix trailing commas before ] or } (happens when a notice is removed from
  1236→    // between the last value and the closing brace, e.g. {"k":"v",Notice:...} → {"k":"v",})
  1237→    .replace(/,\s*([}\]])/g, '$1')
  1238→    // Step 5: Fix missing commas between objects: }{ should be },{
  1239→    .replace(/}\s*{/g, '},{')
  1240→    // Step 6: Fix double commas (happens when a notice is removed from between two commas)
  1241→    .replace(/,\s*,/g, ',')
  1242→
  1243→  return cleaned.trim()
  1244→}

  1246→/**
  1247→ * Safely parse JSON from Air Intra API response.
  1248→ * Handles cases where the API returns:
  1249→ * - PHP notices/warnings BEFORE, AFTER, or INSIDE the JSON array
  1250→ * - HTML error pages instead of JSON
  1251→ * - Proper API error objects with error_id
  1252→ */
  1253→async function safeParseAirIntraResponse(res: Response): Promise<{ data: any; error: string | null }> {
  1254→  const rawText = await res.text()
  1255→
  1256→  // Step 1: Strip PHP notices from the entire response
  1257→  const text = stripPhpNotices(rawText)
  1258→
  1259→  // Step 2: Find the start of JSON ({ or [)
  1260→  let jsonStart = -1
  1261→  for (let i = 0; i < text.length; i++) {
  1262→    const ch = text[i]
  1263→    if (ch === '{' || ch === '[') {
  1264→      jsonStart = i
  1265→      break
  1266→    }
  1267→  }
  1268→
  1269→  if (jsonStart === -1) {
  1270→    const cleanText = text.substring(0, 300)
  1271→    return { data: null, error: `La API no devolvió datos JSON. ${cleanText || 'Respuesta vacía'}` }
  1272→  }
  1273→
  1274→  const jsonText = text.substring(jsonStart)
  1275→
  1276→  // Step 3: Try direct parse first
  1277→  try {
  1278→    const data = JSON.parse(jsonText)
  1279→    // Check for API-level errors
  1280→    if (data && typeof data === 'object' && !Array.isArray(data) && data.error_id) {
  1281→      if (data.error_id === 401) {
  1282→        return { data: null, error: `Token expirado o inválido. ${data.error_detail || ''}` }
  1283→      }
  1284→      if (data.error_id === 403) {
  1285→        return { data: null, error: `Demasiadas solicitudes. La API de Air Intra requiere esperar 5 minutos entre ciclos de consulta. ${data.error_detail || ''}` }
  1286→      }
  1287→      return { data: null, error: `Error API Air Intra (${data.error_id}): ${data.error_name || ''} - ${data.error_detail || ''}` }
  1288→    }
  1289→    return { data, error: null }
  1290→  } catch (firstError: any) {
  1291→    console.log(`[Air Intra] First JSON parse failed (${jsonText.length} chars): ${firstError.message}`)
  1292→  }
  1293→
  1294→  // Step 4: If direct parse failed, try more aggressive cleanup
  1295→  let aggressiveClean = jsonText
  1296→    // Remove any remaining HTML tags
  1297→    .replace(/<[^>]*>/g, '')
  1298→    // Fix double commas (,,) that can appear after removing notices
  1299→    .replace(/,\s*,/g, ',')
  1300→    // Fix missing commas between objects: }{ should be },{
  1301→    .replace(/}\s*{/g, '},{')
  1302→    // Fix trailing commas before ] or }
  1303→    .replace(/,\s*([}\]])/g, '$1')
  1304→
  1305→  try {
  1306→    const data = JSON.parse(aggressiveClean)
  1307→    if (data && typeof data === 'object' && !Array.isArray(data) && data.error_id) {
  1308→      if (data.error_id === 401) {
  1309→        return { data: null, error: `Token expirado o inválido. ${data.error_detail || ''}` }
  1310→      }
  1311→      if (data.error_id === 403) {
  1312→        return { data: null, error: `Demasiadas solicitudes. ${data.error_detail || ''}` }
  1313→      }
  1314→      return { data: null, error: `Error API Air Intra (${data.error_id}): ${data.error_name || ''}` }
  1315→    }
  1316→    console.log('[Air Intra] Aggressive cleanup parse succeeded')
  1317→    return { data, error: null }
  1318→  } catch (secondError: any) {
  1319→    console.log(`[Air Intra] Aggressive cleanup parse also failed: ${secondError.message}`)
  1320→
  1321→    // Try to find the error position for debugging
  1322→    const posMatch = secondError.message.match(/position\s+(\d+)/i)
  1323→    if (posMatch) {
  1324→      const pos = parseInt(posMatch[1])
  1325→      const context = aggressiveClean.substring(Math.max(0, pos - 80), pos + 80)
  1326→      return { data: null, error: `Error JSON en posición ${pos}. Contexto: ...${context}...` }
  1327→    }
  1328→
  1329→    const preview = aggressiveClean.substring(0, 200)
  1330→    return { data: null, error: `No se pudo interpretar la respuesta JSON (${aggressiveClean.length} chars). Preview: ${preview}` }
  1331→  }
  1332→}

  1334→/**
  1335→ * Parse a potentially corrupted Air Intra JSON array by extracting individual
  1336→ * product objects using regex. This is the ultimate fallback when the entire
  1337→ * JSON array can't be parsed due to PHP notices injected between/inside objects.
  1338→ * Returns an array of successfully parsed product objects.
  1339→ */
  1340→function extractProductsFromCorruptedJson(text: string): any[] {
  1341→  const products: any[] = []
  1342→
  1343→  // Find all top-level JSON objects in the text
  1344→  // We look for patterns like {"codigo": ...} which are product objects
  1345→  // We track brace depth to find complete objects
  1346→  let i = 0
  1347→  while (i < text.length) {
  1348→    // Find the start of a JSON object
  1349→    if (text[i] !== '{') {
  1350→      i++
  1351→      continue
  1352→    }
  1353→
  1354→    // Try to parse a complete object starting at position i
  1355→    let depth = 0
  1356→    let inStr = false
  1357→    let esc = false
  1358→    let objEnd = -1
  1359→
  1360→    for (let j = i; j < text.length; j++) {
  1361→      const ch = text[j]
  1362→
  1363→      if (esc) {
  1364→        esc = false
  1365→        continue
  1366→      }
  1367→      if (ch === '\\' && inStr) {
  1368→        esc = true
  1369→        continue
  1370→      }
  1371→      if (ch === '"') {
  1372→        inStr = !inStr
  1373→        continue
  1374→      }
  1375→      if (inStr) continue
  1376→
  1377→      if (ch === '{') depth++
  1378→      else if (ch === '}') {
  1379→        depth--
  1380→        if (depth === 0) {
  1381→          objEnd = j
  1382→          break
  1383→        }
  1384→      }
  1385→    }
  1386→
  1387→    if (objEnd === -1) {
  1388→      i++
  1389→      continue
  1390→    }
  1391→
  1392→    const objText = text.substring(i, objEnd + 1)
  1393→
  1394→    // Quick check: skip objects that are clearly not products
  1395→    // (e.g., error objects, impuesto_iva fragments, etc.)
  1396→    if (objText.includes('"codigo"') || objText.includes('"codiart"')) {
  1397→      try {
  1398→        const obj = JSON.parse(objText)
  1399→        products.push(obj)
  1400→      } catch {
  1401→        // Object itself is corrupted, skip it
  1402→      }
  1403→    }
  1404→
  1405→    i = objEnd + 1
  1406→  }
  1407→
  1408→  return products
  1409→}
