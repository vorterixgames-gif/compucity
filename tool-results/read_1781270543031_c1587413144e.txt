    30→export default function CheckoutPage() {

    35→    name: '',
    36→    dni: '',
    37→    email: '',
    38→    phone: '',
    39→    address: '',
    40→    city: '',
    41→    province: '',

    58→            name: c.name || prev.name,

    60→            phone: c.phone || prev.phone,
    61→            dni: c.dni || prev.dni,
    62→            address: c.address || prev.address,
    63→            city: c.city || prev.city,
    64→            province: c.province || prev.province,
    65→            postalCode: c.postalCode || prev.postalCode,
    66→          }))
    67→        }
    68→      } catch {
    69→        // Not logged in, that's fine
    70→      }
    71→    }

    91→        carrierName: 'Retiro en local',
    92→        service: 'retiro',
    93→        serviceName: 'Retiro en CompuCity',

   127→        setSelectedQuote(null)

   165→      message += `🏷️ *Cupón ${appliedCoupon.code}: -${formatPrice(couponDiscount)}*\n`
   166→    }
   167→    if (shippingCost > 0) {
   168→      message += `📦 *Envío: ${formatPrice(shippingCost)}*\n`
   169→    }

   172→      message += `\n📝 *Notas:* ${customerData.notes}\n`

   177→  }

   179→  const [sending, setSending] = useState(false)
   180→  const [orderError, setOrderError] = useState('')
   181→
   182→  // Login modal state
   183→  const [showLoginModal, setShowLoginModal] = useState(false)
   184→  const [loginEmail, setLoginEmail] = useState('')
   185→  const [loginPassword, setLoginPassword] = useState('')
   186→  const [showPassword, setShowPassword] = useState(false)
   187→  const [loginError, setLoginError] = useState('')
   188→  const [loginLoading, setLoginLoading] = useState(false)
   189→
   190→  // Post-order account creation state
   191→  const [showAccountOffer, setShowAccountOffer] = useState(false)
   192→  const [createAccountPassword, setCreateAccountPassword] = useState('')
   193→  const [createAccountLoading, setCreateAccountLoading] = useState(false)

   198→

   210→        setLoginError(data.error || 'Credenciales inválidas')
   211→        setLoginLoading(false)
   212→        return

   215→      const c = data.customer as LoggedInCustomer
   216→      setLoggedInCustomer(c)
   217→      setCustomerData(prev => ({
   218→        ...prev,
   219→        name: c.name || prev.name,
   220→        email: c.email || prev.email,
   221→        phone: c.phone || prev.phone,

   273→    }
   274→  }

   280→      // 1. Guardar pedido en la base de datos
   281→      const orderRes = await fetch('/api/orders', {
   282→        method: 'POST',
   283→        headers: { 'Content-Type': 'application/json' },
   284→        body: JSON.stringify({
   285→          customerName: customerData.name,
   286→          customerDni: customerData.dni,
   287→          customerEmail: customerData.email,
   288→          customerPhone: customerData.phone,
   289→          customerId: loggedInCustomer?.id || null,
   290→          shippingAddress: customerData.address,
   291→          shippingCity: customerData.city,
   292→          shippingProvince: customerData.province,
   293→          shippingZip: customerData.postalCode,
   294→          shippingMethod: shippingMethod === 'retiro' ? 'retiro' : 'envio',
   295→          shippingCost,
   296→          shippingDetails: shippingMethod === 'envio' && selectedQuote
   297→            ? JSON.stringify({ carrier: selectedQuote.carrier, carrierName: selectedQuote.carrierName, service: selectedQuote.service, serviceName: selectedQuote.serviceName, price: selectedQuote.price, estimatedDays: selectedQuote.estimatedDays, description: selectedQuote.description })
   298→            : JSON.stringify({ method: 'retiro', description: 'Retiro en local - Av. Sarmiento 462, La Falda, Córdoba' }),
   299→          notes: customerData.notes,
   300→          couponCode: appliedCoupon?.code || null,
   301→          couponDiscount: couponDiscount > 0 ? couponDiscount : null,
   302→          items: items.map(item => ({
   303→            productId: item.id,
   304→            name: item.name,
   305→            price: item.price,
   306→            quantity: item.quantity,
   307→          })),
   308→          total: grandTotal,
   309→        }),
   310→      })
   311→
   312→      const orderData = await orderRes.json()
   313→
   314→      if (!orderRes.ok) {
   315→        setOrderError(orderData.error || 'Error al guardar el pedido')
   316→        setSending(false)
   317→        return
   318→      }
   319→
   320→      // 2. Abrir WhatsApp con el mensaje
   321→      const url = generateWhatsAppMessage()
   322→      window.open(url, '_blank')
   323→      clearCart()
   324→
   325→      // 3. Si no está logueado y tiene email, ofrecer crear cuenta
   326→      if (!loggedInCustomer && customerData.email) {
   327→        setShowAccountOffer(true)
   328→      }
   329→      setOrderCompleted(true)
   330→    } catch (err) {
   331→      setOrderError('Error de conexión. Intentá de nuevo.')
   332→    } finally {
   333→      setSending(false)
   334→    }
   335→  }
   336→
   337→  // Pantalla de pedido completado

   339→    return (
   340→      <div className="max-w-2xl mx-auto px-4 py-16 text-center">

   343→        <p className="text-gray-500 mb-6">
   344→          Tu pedido fue enviado por WhatsApp. Un vendedor se va a contactar a la brevedad para confirmar y coordinar el pago y la entrega.
   345→        </p>
   346→
   347→        {/* Ofrecer crear cuenta si no está logueado y tiene email */}
   348→        {showAccountOffer && !accountCreated && (
   349→          <div className="relative mt-6 p-6 bg-compucity-green-50 border border-compucity-green-100 rounded-xl text-left max-w-md mx-auto">
   350→            <div className="flex items-center gap-2 mb-3">
   351→              <UserPlus className="h-5 w-5 text-compucity-green" />
   352→              <h3 className="font-semibold text-compucity-green-900">¿Querés crear una cuenta?</h3>
   353→            </div>
   354→            <p className="text-sm text-compucity-green-700 mb-4">
   355→              Con una cuenta podés seguir tus pedidos, guardar tus datos para la próxima compra y acceder a ofertas exclusivas.
   356→            </p>
   357→            {createAccountError && (
   358→              <div className="mb-3 p-2 bg-red-50 border border-red-200 rounded-lg text-sm text-red-700">{createAccountError}</div>
   359→            )}
   360→            <div className="space-y-3">
   361→              {/* Honeypot - campo oculto para bots, los humanos no lo ven */}
   362→              <div className="absolute opacity-0 h-0 w-0 overflow-hidden" aria-hidden="true" tabIndex={-1}>
   363→                <label htmlFor="website_hp">No completar este campo</label>
   364→                <input
   365→                  id="website_hp"
   366→                  type="text"
   367→                  value={honeypot}
   368→                  onChange={(e) => setHoneypot(e.target.value)}

   370→                  autoComplete="off"

   376→                  type="password"
   377→                  value={createAccountPassword}
   378→                  onChange={(e) => setCreateAccountPassword(e.target.value)}
   379→                  placeholder="Mínimo 6 caracteres"
   380→                  className="w-full border rounded-lg px-4 py-2 text-sm focus:outline-none focus:border-compucity-green"
   381→                />
   382→              </div>
   383→              <button
   384→                onClick={handleCreateAccount}
   385→                disabled={createAccountPassword.length < 6 || createAccountLoading}
   386→                className="w-full flex items-center justify-center gap-2 py-2.5 bg-compucity-green text-white font-medium rounded-lg hover:bg-compucity-green-dark disabled:bg-gray-300 disabled:cursor-not-allowed transition"
   387→              >
   388→                {createAccountLoading ? <Loader2 className="h-4 w-4 animate-spin" /> : <UserPlus className="h-4 w-4" />}
   389→                {createAccountLoading ? 'Creando cuenta...' : 'Crear mi cuenta'}
   390→              </button>
   391→              <button
   392→                onClick={() => setShowAccountOffer(false)}
   393→                className="w-full text-sm text-gray-400 hover:text-gray-600 transition"
   394→              >
   395→                No, gracias
   396→              </button>
   397→            </div>
   398→          </div>
   399→        )}
   400→
   401→        {accountCreated && (
   402→          <div className="mt-6 p-4 bg-green-50 border border-green-200 rounded-xl max-w-md mx-auto">
   403→            <CheckCircle className="h-6 w-6 text-green-600 mx-auto mb-2" />
   404→            <p className="font-semibold text-green-800">¡Cuenta creada con éxito!</p>

   408→
   409→        <Link href="/" className="inline-block mt-6 px-6 py-3 bg-compucity-green text-white rounded-lg hover:bg-compucity-green-dark transition">
   410→          Volver a la tienda

   412→      </div>

   415→
   416→  if (items.length === 0 && !orderCompleted) {
   417→    return (
   418→      <div className="max-w-2xl mx-auto px-4 py-16 text-center">
   419→        <Package className="h-16 w-16 text-gray-300 mx-auto mb-4" />
   420→        <h1 className="text-2xl font-bold mb-2">Tu carrito está vacío</h1>
   421→        <p className="text-gray-500 mb-6">Agregá productos al carrito para hacer un pedido.</p>
   422→        <Link href="/" className="px-6 py-3 bg-compucity-green text-white rounded-lg hover:bg-compucity-green-dark transition">
   423→          Ver productos
   424→        </Link>
   425→      </div>
   426→    )
   427→  }
   428→
   429→  const canProceedFromData = customerData.name && customerData.phone
   430→  const canSendOrder = customerData.name && customerData.phone && (
   431→    shippingMethod === 'retiro' || (
   432→      customerData.address && customerData.city && customerData.province && customerData.postalCode && selectedQuote
   433→    )
   434→  )
   435→
   436→  return (
   437→    <div className="max-w-4xl mx-auto px-4 py-8">
   438→      <h1 className="text-2xl font-bold text-gray-900 mb-6">Finalizar Pedido</h1>
   439→
   440→      {/* Steps indicator */}
   441→      <div className="flex items-center gap-2 mb-8">

   445→          const Icon = icons[i]

   447→          const isDone = ['data', 'shipping'].indexOf(step) > i
   448→          return (
   449→            <button key={s} onClick={() => { if (isDone) setStep(s) }} className="flex items-center gap-2">
   450→              <div className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-medium ${isActive ? 'bg-compucity-green text-white' : isDone ? 'bg-green-500 text-white' : 'bg-gray-200 text-gray-500'}`}>
   451→                <Icon className="h-4 w-4" />

   453→              <span className={`text-sm ${isActive ? 'text-compucity-green font-medium' : 'text-gray-500'}`}>{labels[i]}</span>

   455→            </button>
   456→          )
   457→        })}
   458→      </div>
   459→
   460→      <div className="grid md:grid-cols-3 gap-8">
   461→        {/* Form */}

   465→              {/* Login prompt - solo si NO está logueado */}
   466→              {!loggedInCustomer && (

   469→                    <LogIn className="h-4 w-4 text-compucity-green shrink-0" />

   472→                  <button
   473→                    onClick={() => { setShowLoginModal(true); setLoginError('') }}
   474→                    className="text-sm font-semibold text-compucity-green hover:text-compucity-green-dark transition"
   475→                  >
   476→                    Iniciá sesión
   477→                  </button>
   478→                </div>
   479→              )}
   480→
   481→              {/* Logged in indicator */}
   482→              {loggedInCustomer && (
   483→                <div className="flex items-center gap-2 p-3 bg-green-50 border border-green-200 rounded-lg">
   484→                  <CheckCircle className="h-4 w-4 text-green-600 shrink-0" />
   485→                  <span className="text-sm text-green-800">
   486→                    Sesión iniciada como <strong>{loggedInCustomer.name}</strong> — tus datos se cargaron automáticamente
   487→                  </span>
   488→                </div>
   489→              )}
   490→
   491→              <div className="bg-white border rounded-lg p-6 space-y-4">
   492→                <h2 className="text-lg font-semibold">Tus datos</h2>
   493→                <p className="text-sm text-gray-500">Completá tus datos para que los vendedores puedan contactarte y coordinar la entrega.</p>
   494→              <div className="grid md:grid-cols-2 gap-4">
   495→                <div>
   496→                  <label className="block text-sm font-medium text-gray-700 mb-1">Nombre completo *</label>
   497→                  <div className="relative">
   498→                    <User className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   499→                    <input
   500→                      type="text"
   501→                      value={customerData.name}
   502→                      onChange={(e) => setCustomerData({ ...customerData, name: e.target.value })}
