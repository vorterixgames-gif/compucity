   211→/**
   212→ * Build a category lookup from the database.
   213→ * Returns slug -> id map, name -> id map, and parent info for subcategory logic.
   214→ */
   215→async function buildCategoryLookup(): Promise<{
   216→  slugToId: Record<string, string>
   217→  nameToId: Record<string, string>
   218→  idToParentId: Record<string, string | null>
   219→  parentSlugToChildSlugs: Record<string, string[]>
   220→}> {
   221→  const result = await db.execute('SELECT id, name, slug, parentId FROM categories')
   222→  const slugToId: Record<string, string> = {}
   223→  const nameToId: Record<string, string> = {}
   224→  const idToParentId: Record<string, string | null> = {}
   225→  const parentSlugToChildSlugs: Record<string, string[]> = {}
   226→  for (const row of result.rows as any[]) {
   227→    if (row.slug) slugToId[row.slug] = row.id
   228→    if (row.name) nameToId[row.name.toLowerCase()] = row.id
   229→    idToParentId[row.id] = row.parentId || null
   230→    if (row.parentId) {
   231→      // Find parent slug
   232→      const parentRow = (result.rows as any[]).find((r: any) => r.id === row.parentId)
   233→      if (parentRow?.slug) {
   234→        if (!parentSlugToChildSlugs[parentRow.slug]) parentSlugToChildSlugs[parentRow.slug] = []
   235→        parentSlugToChildSlugs[parentRow.slug].push(row.slug)
   236→      }
   237→    }
   238→  }
   239→  return { slugToId, nameToId, idToParentId, parentSlugToChildSlugs }
   240→}

   242→/**
   243→ * Build supplier category mappings from the database.
   244→ * Returns supplierCategory -> storeCategoryId map for a given supplier.
   245→ */
   246→async function buildSupplierMappingLookup(supplierId: string): Promise<Record<string, string>> {
   247→  const result = await db.execute({
   248→    sql: 'SELECT supplierCategory, storeCategoryId FROM supplier_category_mappings WHERE supplierId = ?',
   249→    args: [supplierId],
   250→  })
   251→  const map: Record<string, string> = {}
   252→  for (const row of result.rows as any[]) {
   253→    map[row.supplierCategory] = row.storeCategoryId
   254→  }
   255→  return map
   256→}

   258→/**

   260→ * 1. Supplier category mapping (if available)
   261→ * 2. Keyword matching with slug-based lookup
   262→ * 3. Subcategory refinement: if the matched category is a parent with subcategories,
   263→ *    use subcategory rules to find the best subcategory match
   264→ * 4. Default: null (no category)
   265→ */
   266→function mapProductToCategory(
   267→  productName: string,
   268→  supplierCategory: string | null,
   269→  supplierMappings: Record<string, string>,
   270→  slugToId: Record<string, string>,

   275→  let matchedCategoryId: string | null = null
   276→  let method = 'none'
   277→

   279→  if (supplierCategory && supplierMappings[supplierCategory]) {
   280→    matchedCategoryId = supplierMappings[supplierCategory]
   281→    method = 'mapping'
   282→  }
   283→
   284→  // 2. Keyword matching (only if no mapping match)

  1208→/**
  1209→ * Strip PHP notices/warnings/errors from Air Intra API response text.
  1210→ * Air Intra's PHP server injects HTML-formatted PHP notices like:
  1211→ *   <br /> <b>Notice</b>: Undefined property: stdClass::$estado in /path/file.php on line 54
  1212→ * These can appear BEFORE, AFTER, or INSIDE the JSON array (between product objects).
  1213→ * This function removes them while preserving the actual JSON data.
  1214→ */
  1215→function stripPhpNotices(text: string): string {
  1216→  let cleaned = text
  1217→    // Remove complete PHP notice/warning/error blocks (HTML format with <b> tags)
  1218→    // Pattern: optional <br />, then <b>Type</b>: message in /path/file.php on line NNN
  1219→    .replace(/(?:<br\s*\/?>\s*)?<b>(?:Notice|Warning|Fatal error|Parse error|Deprecated)<\/b>:\s*.*?on line \d+\s*/gis, '')
  1220→    // Remove plain-text PHP notices (without HTML tags)
  1221→    .replace(/(?:^|\n)\s*(?:Notice|Warning|Fatal error|Parse error|Deprecated):\s*.*?on line \d+\s*/gis, '')
  1222→    // Remove any remaining standalone <br /> tags
  1223→    .replace(/<br\s*\/?>\s*/gi, '')
  1224→    // Remove leftover <b> or </b> tags
  1225→    .replace(/<\/?b>/gi, '')
  1226→    // Fix trailing commas before ] or } (can happen after removing notices)
  1227→    .replace(/,\s*([}\]])/g, '$1')
  1228→
  1229→  return cleaned.trim()
  1230→}

  1232→/**
  1233→ * Safely parse JSON from Air Intra API response.
  1234→ * Handles cases where the API returns:
  1235→ * - PHP notices/warnings BEFORE, AFTER, or INSIDE the JSON array
  1236→ * - HTML error pages instead of JSON
  1237→ * - Proper API error objects with error_id
  1238→ */
  1239→async function safeParseAirIntraResponse(res: Response): Promise<{ data: any; error: string | null }> {
  1240→  const rawText = await res.text()
  1241→
  1242→  // Step 1: Strip PHP notices from the entire response
  1243→  const text = stripPhpNotices(rawText)
  1244→
  1245→  // Step 2: Find the start of JSON ({ or [)
  1246→  let jsonStart = -1
  1247→  for (let i = 0; i < text.length; i++) {
  1248→    const ch = text[i]
  1249→    if (ch === '{' || ch === '[') {
  1250→      jsonStart = i
  1251→      break
  1252→    }
  1253→  }
  1254→
  1255→  if (jsonStart === -1) {
  1256→    const cleanText = text.substring(0, 300)
  1257→    return { data: null, error: `La API no devolvió datos JSON. ${cleanText || 'Respuesta vacía'}` }
  1258→  }
  1259→
  1260→  const jsonText = text.substring(jsonStart)
  1261→
  1262→  // Step 3: Try direct parse first
  1263→  try {
  1264→    const data = JSON.parse(jsonText)
  1265→    // Check for API-level errors
  1266→    if (data && typeof data === 'object' && !Array.isArray(data) && data.error_id) {
  1267→      if (data.error_id === 401) {
  1268→        return { data: null, error: `Token expirado o inválido. ${data.error_detail || ''}` }
  1269→      }
  1270→      if (data.error_id === 403) {
  1271→        return { data: null, error: `Demasiadas solicitudes. La API de Air Intra requiere esperar 5 minutos entre ciclos de consulta. ${data.error_detail || ''}` }
  1272→      }
  1273→      return { data: null, error: `Error API Air Intra (${data.error_id}): ${data.error_name || ''} - ${data.error_detail || ''}` }
  1274→    }
  1275→    return { data, error: null }
  1276→  } catch (firstError: any) {
  1277→    console.log(`[Air Intra] First JSON parse failed (${jsonText.length} chars): ${firstError.message}`)
  1278→  }
  1279→
  1280→  // Step 4: If direct parse failed, try more aggressive cleanup
  1281→  let aggressiveClean = jsonText
  1282→    // Remove any remaining HTML tags
  1283→    .replace(/<[^>]*>/g, '')
  1284→    // Fix double commas (,,) that can appear after removing notices
  1285→    .replace(/,\s*,/g, ',')
  1286→    // Fix missing commas between objects: }{ should be },{
  1287→    .replace(/}\s*{/g, '},{')
  1288→    // Fix trailing commas before ] or }
  1289→    .replace(/,\s*([}\]])/g, '$1')
  1290→
  1291→  try {
  1292→    const data = JSON.parse(aggressiveClean)
  1293→    if (data && typeof data === 'object' && !Array.isArray(data) && data.error_id) {
  1294→      if (data.error_id === 401) {
  1295→        return { data: null, error: `Token expirado o inválido. ${data.error_detail || ''}` }
  1296→      }
  1297→      if (data.error_id === 403) {
  1298→        return { data: null, error: `Demasiadas solicitudes. ${data.error_detail || ''}` }
  1299→      }
  1300→      return { data: null, error: `Error API Air Intra (${data.error_id}): ${data.error_name || ''}` }
  1301→    }
  1302→    console.log('[Air Intra] Aggressive cleanup parse succeeded')
  1303→    return { data, error: null }
  1304→  } catch (secondError: any) {
  1305→    console.log(`[Air Intra] Aggressive cleanup parse also failed: ${secondError.message}`)
  1306→
  1307→    // Try to find the error position for debugging
  1308→    const posMatch = secondError.message.match(/position\s+(\d+)/i)
  1309→    if (posMatch) {
  1310→      const pos = parseInt(posMatch[1])
  1311→      const context = aggressiveClean.substring(Math.max(0, pos - 80), pos + 80)
  1312→      return { data: null, error: `Error JSON en posición ${pos}. Contexto: ...${context}...` }
  1313→    }
  1314→
  1315→    const preview = aggressiveClean.substring(0, 200)
  1316→    return { data: null, error: `No se pudo interpretar la respuesta JSON (${aggressiveClean.length} chars). Preview: ${preview}` }
  1317→  }
  1318→}

  1320→/**
  1321→ * Parse a potentially corrupted Air Intra JSON array by extracting individual
  1322→ * product objects using regex. This is the ultimate fallback when the entire
  1323→ * JSON array can't be parsed due to PHP notices injected between/inside objects.
  1324→ * Returns an array of successfully parsed product objects.
  1325→ */
  1326→function extractProductsFromCorruptedJson(text: string): any[] {
  1327→  const products: any[] = []
  1328→
  1329→  // Find all top-level JSON objects in the text
  1330→  // We look for patterns like {"codigo": ...} which are product objects
  1331→  // We track brace depth to find complete objects
  1332→  let i = 0
  1333→  while (i < text.length) {
  1334→    // Find the start of a JSON object
  1335→    if (text[i] !== '{') {
  1336→      i++
  1337→      continue
  1338→    }
  1339→
  1340→    // Try to parse a complete object starting at position i
  1341→    let depth = 0
  1342→    let inStr = false
  1343→    let esc = false
  1344→    let objEnd = -1
  1345→
  1346→    for (let j = i; j < text.length; j++) {
  1347→      const ch = text[j]
  1348→
  1349→      if (esc) {
  1350→        esc = false
  1351→        continue
  1352→      }
  1353→      if (ch === '\\' && inStr) {
  1354→        esc = true
  1355→        continue
  1356→      }
  1357→      if (ch === '"') {
  1358→        inStr = !inStr
  1359→        continue
  1360→      }
  1361→      if (inStr) continue
  1362→
  1363→      if (ch === '{') depth++
  1364→      else if (ch === '}') {
  1365→        depth--
  1366→        if (depth === 0) {
  1367→          objEnd = j
  1368→          break
  1369→        }
  1370→      }
  1371→    }
  1372→
  1373→    if (objEnd === -1) {
  1374→      i++
  1375→      continue
  1376→    }
  1377→
  1378→    const objText = text.substring(i, objEnd + 1)
  1379→
  1380→    // Quick check: skip objects that are clearly not products
  1381→    // (e.g., error objects, impuesto_iva fragments, etc.)
  1382→    if (objText.includes('"codigo"') || objText.includes('"codiart"')) {
  1383→      try {
  1384→        const obj = JSON.parse(objText)
  1385→        products.push(obj)
