   193→/**
   194→ * Parse shipping details from JSON or legacy notes format
   195→ */
   196→function parseShippingDetails(order: OrderData): ShippingDetailsJSON | null {
   197→  // Try new JSON format first
   198→  if (order.shippingDetails) {
   199→    try {
   200→      return JSON.parse(order.shippingDetails)
   201→    } catch {
   202→      // Not valid JSON, fall through
   203→    }
   204→  }
   205→
   206→  // Try legacy format from notes (e.g. "[Envío: Andreani Estándar · 3-7 días · Andreani]")
   207→  if (order.notes) {
   208→    const match = order.notes.match(/\[Envío: (.+?)\]/)
   209→    if (match) {
   210→      return { description: match[1], serviceName: match[1] }
   211→    }
   212→  }
   213→
   214→  return null
   215→}

   217→/**
   218→ * Get tracking URL based on tracking number format
   219→ */
   220→function getTrackingUrl(trackingNumber: string): { url: string; carrier: string } | null {
   221→  if (!trackingNumber) return null
   222→  const tn = trackingNumber.trim().toUpperCase()
   223→
   224→  // Andreani tracking numbers are typically numeric, 10-14 digits
   225→  if (/^\d{10,14}$/.test(tn)) {
   226→    return { url: `https://www.andreani.com/Centralizador-de-envios/Consulta?numero=${tn}`, carrier: 'Andreani' }
   227→  }
   228→
   229→  // Correo Argentino tracking starts with specific patterns
   230→  if (/^[A-Z]{2}\d{9}[A-Z]{2}$/.test(tn) || tn.startsWith('CA') || tn.startsWith('RA')) {
   231→    return { url: `https://www.correoargentino.com.ar/formularios/e-commerce`, carrier: 'Correo Argentino' }
   232→  }
   233→
   234→  // OCA tracking
   235→  if (tn.startsWith('OCA') || /^[A-Z]{3}\d+$/.test(tn)) {
   236→    return { url: `https://www.oca.com.ar/Seguimiento/Envio/${tn}`, carrier: 'OCA' }
   237→  }
   238→
   239→  // Generic — return the number without a specific URL
   240→  return { url: '', carrier: 'Transporte' }
   241→}
