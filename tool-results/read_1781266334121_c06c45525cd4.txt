   110→   110→function isExcludedFromBuilder(slot: string, name: string): boolean {
   111→   111→  const upper = name.toUpperCase()
   112→   112→  const includePatterns = BUILDER_INCLUDE_PATTERNS[slot]
   113→   113→  if (includePatterns && includePatterns.length > 0) {
   114→   114→    if (!includePatterns.some(p => upper.includes(p.toUpperCase()))) return true
   115→   115→  }
   116→   116→  const excludePatterns = BUILDER_EXCLUDE_PATTERNS[slot]
   117→   117→  if (excludePatterns) {
   118→   118→    if (excludePatterns.some(p => upper.includes(p.toUpperCase()))) return true
   119→   119→  }
   120→   120→  return false
   121→   121→}

   145→   145→function parseLlmJson(raw: string): any | null {
   146→   146→  try { return JSON.parse(raw) } catch {}
   147→   147→  const codeBlockMatch = raw.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/)
   148→   148→  if (codeBlockMatch) { try { return JSON.parse(codeBlockMatch[1].trim()) } catch {} }
   149→   149→  const braceStart = raw.indexOf('{')
   150→   150→  const braceEnd = raw.lastIndexOf('}')
   151→   151→  if (braceStart !== -1 && braceEnd > braceStart) { try { return JSON.parse(raw.substring(braceStart, braceEnd + 1)) } catch {} }
   152→   152→  return null
   153→   153→}

   168→   168→      const ids: string[] = []
   169→   169→
   170→   170→      for (const slug of slugs) {
   171→   171→        if (allCategoryIds.has(slug)) {
   172→   172→          ids.push(...allCategoryIds.get(slug)!)
   173→   173→          continue
   174→   174→        }
   175→   175→        const catResult = await db.execute({
   176→   176→          sql: 'SELECT id FROM categories WHERE slug = ?',

   179→   179→        const catRows = catResult.rows as any[]
   180→   180→        if (catRows.length === 0) {
   181→   181→          console.warn(`[pc-assistant] Category slug not found: ${slug}`)

   184→   184→        }
   185→   185→
   186→   186→        const catId = catRows[0].id
   187→   187→        const subResult = await db.execute({
   188→   188→          sql: 'SELECT id FROM categories WHERE parentId = ?',

   192→   192→        allCategoryIds.set(slug, catIds)

   214→   214→      slotProducts.set(slotDef.slot, products.map(p => ({

   239→   239→
   240→   240→    if (allProducts.length === 0) {
   241→   241→      console.warn('[pc-assistant] No products found in any category')

   244→   244→

   291→   291→
   292→   292→  return slotProducts
   293→   293→}

   295→   295→// ============================================
   296→   296→// Pick Best Product for a Slot
   297→   297→// ============================================
   298→   298→
   299→   299→function pickProductForSlot(
   300→   300→  products: ProductWithArsPrice[],
   301→   301→  idealMin: number,
   302→   302→  idealMax: number,
   303→   303→  compatFilter?: (p: ProductWithArsPrice) => boolean
   304→   304→): ProductWithArsPrice | null {
   305→   305→  if (products.length === 0) return null
   306→   306→
   307→   307→  // Apply compatibility filter if provided
   308→   308→  let pool = compatFilter ? products.filter(compatFilter) : products
   309→   309→
   310→   310→  // If compatibility filter removed everything, fall back to unfiltered
   311→   311→  if (pool.length === 0) {
   312→   312→    console.warn('[pc-assistant] Compatibility filter removed all products, using unfiltered pool')
   313→   313→    pool = products
   314→   314→  }
   315→   315→
   316→   316→  if (pool.length === 0) return null
   317→   317→
   318→   318→  // Score products: prefer products near the midpoint of the ideal range
   319→   319→  // but allow significant flexibility (up to 2x idealMax for expensive components)
   320→   320→  const midPrice = (idealMin + idealMax) / 2
   321→   321→
   322→   322→  // First try: products within ideal range
   323→   323→  let candidates = pool.filter(p => {
   324→   324→    const price = p.arsComparePrice || p.arsPrice
   325→   325→    return price >= idealMin * 0.3 && price <= idealMax * 1.5
   326→   326→  })
   327→   327→
   328→   328→  // Second try: just cap at 2x max, no min
   329→   329→  if (candidates.length === 0) {
   330→   330→    candidates = pool.filter(p => {
   331→   331→      const price = p.arsComparePrice || p.arsPrice
   332→   332→      return price <= idealMax * 2
   333→   333→    })
   334→   334→  }
   335→   335→
   336→   336→  // Third try: any product, sorted by price
   337→   337→  if (candidates.length === 0) {
   338→   338→    candidates = pool
   339→   339→  }
   340→   340→
   341→   341→  if (candidates.length === 0) return null
   342→   342→
   343→   343→  // Score: prefer products closest to the midpoint of the allocation
   344→   344→  // This fills the budget better - midPrice is the sweet spot for value
   345→   345→  candidates.sort((a, b) => {
   346→   346→    const priceA = a.arsComparePrice || a.arsPrice
   347→   347→    const priceB = b.arsComparePrice || b.arsPrice
   348→   348→    const scoreA = Math.abs(priceA - midPrice)
   349→   349→    const scoreB = Math.abs(priceB - midPrice)
   350→   350→    return scoreA - scoreB
   351→   351→  })
   352→   352→
   353→   353→  return candidates[0]
   354→   354→}

   373→   373→  // Track compatibility constraints
   374→   374→  let processorSocket: string | null = null
   375→   375→  let motherboardDdr: string | null = null
   376→   376→  let gpuMinWattage: number | null = null
   377→   377→
   378→   378→  // Calculate price allocations per slot
   379→   379→  const allocations: Record<string, number> = {}
   380→   380→  for (const [slot, pct] of Object.entries(profile)) {
   381→   381→    if (pct === 0) continue
   382→   382→    allocations[slot] = effectiveBudget * pct
   383→   383→  }
   384→   384→
   385→   385→  // ---- Step 1: Processor ----
   386→   386→  const processorProducts = allProducts.get('processor') || []
   387→   387→  const procAlloc = allocations['processor']
   388→   388→  if (!procAlloc || processorProducts.length === 0) {
   389→   389→    console.error(`[pc-assistant] Cannot build: processor alloc=${procAlloc}, products=${processorProducts.length}`)
   390→   390→    return null
   391→   391→  }
   392→   392→
   393→   393→  const processor = pickProductForSlot(processorProducts, procAlloc * 0.4, procAlloc * 1.3)
   394→   394→  if (!processor) {
   395→   395→    console.error('[pc-assistant] Cannot build: no processor selected')
   396→   396→    return null
   397→   397→  }
   398→   398→
   399→   399→  const procCompat = extractCompatibility('processor', processor.name)
   400→   400→  processorSocket = procCompat.socket || null
   401→   401→  debugLog.push(`Processor: ${processor.name} ($${Math.round(processor.arsComparePrice).toLocaleString()} ARS) socket=${processorSocket}`)
   402→   402→
   403→   403→  components.push(makeComponent(processor, 'processor', 'Microprocesador'))
   404→   404→
   405→   405→  // ---- Step 2: Motherboard ----
   406→   406→  const mbProducts = allProducts.get('motherboard') || []
   407→   407→  const mbAlloc = allocations['motherboard']
   408→   408→  if (!mbAlloc || mbProducts.length === 0) {
   409→   409→    console.error(`[pc-assistant] Cannot build: motherboard alloc=${mbAlloc}, products=${mbProducts.length}`)
   410→   410→    return null
   411→   411→  }
   412→   412→
   413→   413→  // Filter by socket compatibility
   414→   414→  const mbCompatFilter = processorSocket
   415→   415→    ? (p: ProductWithArsPrice) => {
   416→   416→        const upper = p.name.toUpperCase()
   417→   417→        const socketPatterns: Record<string, string[]> = {
   418→   418→          AM4: ['AM4', 'B550', 'A520', 'X570', 'B450'],
   419→   419→          AM5: ['AM5', 'B650', 'B850', 'A620', 'X670', 'X870'],
   420→   420→          '1700': ['1700', 'B760', 'H610', 'B660', 'Z690', 'Z790'],
   421→   421→          '1851': ['1851', 'B860', 'Z890', 'H810'],
   422→   422→        }
   423→   423→        const patterns = socketPatterns[processorSocket]
   424→   424→        return patterns ? patterns.some(pat => upper.includes(pat.toUpperCase())) : true
   425→   425→      }
   426→   426→    : undefined
   427→   427→
   428→   428→  const motherboard = pickProductForSlot(mbProducts, mbAlloc * 0.4, mbAlloc * 1.5, mbCompatFilter)
   429→   429→  if (!motherboard) {
   430→   430→    console.error('[pc-assistant] Cannot build: no motherboard selected')
   431→   431→    return null
   432→   432→  }
   433→   433→
   434→   434→  const mbCompat = extractCompatibility('motherboard', motherboard.name)
   435→   435→  motherboardDdr = mbCompat.ddr || null
   436→   436→  debugLog.push(`Motherboard: ${motherboard.name} ($${Math.round(motherboard.arsComparePrice).toLocaleString()} ARS) ddr=${motherboardDdr}`)
   437→   437→
   438→   438→  components.push(makeComponent(motherboard, 'motherboard', 'Motherboard'))
   439→   439→
   440→   440→  // ---- Step 3: RAM ----
   441→   441→  const ramProducts = allProducts.get('ram') || []
   442→   442→  const ramAlloc = allocations['ram']
   443→   443→  if (!ramAlloc || ramProducts.length === 0) {
   444→   444→    console.error(`[pc-assistant] Cannot build: RAM alloc=${ramAlloc}, products=${ramProducts.length}`)
   445→   445→    return null
   446→   446→  }
   447→   447→
   448→   448→  const ramCompatFilter = motherboardDdr
   449→   449→    ? (p: ProductWithArsPrice) => p.name.toUpperCase().includes(motherboardDdr!.toUpperCase())
   450→   450→    : undefined
   451→   451→
   452→   452→  const ram = pickProductForSlot(ramProducts, ramAlloc * 0.3, ramAlloc * 1.5, ramCompatFilter)
   453→   453→  if (!ram) {
   454→   454→    console.error('[pc-assistant] Cannot build: no RAM selected')
   455→   455→    return null
   456→   456→  }
   457→   457→
   458→   458→  debugLog.push(`RAM: ${ram.name} ($${Math.round(ram.arsComparePrice).toLocaleString()} ARS)`)
   459→   459→  components.push(makeComponent(ram, 'ram', 'Memoria RAM'))
   460→   460→
   461→   461→  // ---- Step 4: GPU (optional for some use cases) ----
   462→   462→  if (profile.gpu > 0) {
   463→   463→    const gpuProducts = allProducts.get('gpu') || []
   464→   464→    const gpuAlloc = allocations['gpu']
   465→   465→
   466→   466→    if (gpuAlloc && gpuProducts.length > 0) {
   467→   467→      const gpu = pickProductForSlot(gpuProducts, gpuAlloc * 0.3, gpuAlloc * 1.5)
   468→   468→
   469→   469→      if (gpu) {
   470→   470→        const gpuCompat = extractCompatibility('gpu', gpu.name)
   471→   471→        gpuMinWattage = gpuCompat.gpuTdp ? gpuCompat.gpuTdp * 1.5 + 100 : null
   472→   472→        debugLog.push(`GPU: ${gpu.name} ($${Math.round(gpu.arsComparePrice).toLocaleString()} ARS) wattage_needed=${gpuMinWattage}`)
   473→   473→        components.push(makeComponent(gpu, 'gpu', 'Placa de Video'))
   474→   474→      } else {
   475→   475→        debugLog.push(`GPU: none found in budget range $${Math.round(gpuAlloc * 0.3).toLocaleString()}-$${Math.round(gpuAlloc * 1.5).toLocaleString()}`)
   476→   476→      }
   477→   477→    }
   478→   478→  }
   479→   479→
   480→   480→  // ---- Step 5: SSD ----
   481→   481→  const ssdProducts = allProducts.get('ssd') || []
   482→   482→  const ssdAlloc = allocations['ssd']
   483→   483→
   484→   484→  if (ssdAlloc && ssdProducts.length > 0) {
   485→   485→    const ssd = pickProductForSlot(ssdProducts, ssdAlloc * 0.3, ssdAlloc * 1.5)
   486→   486→    if (ssd) {
   487→   487→      debugLog.push(`SSD: ${ssd.name} ($${Math.round(ssd.arsComparePrice).toLocaleString()} ARS)`)
   488→   488→      components.push(makeComponent(ssd, 'ssd', 'Disco SSD'))
   489→   489→    } else {
   490→   490→      debugLog.push('SSD: none found')
   491→   491→      return null // SSD is required
   492→   492→    }
   493→   493→  } else {
   494→   494→    return null
   495→   495→  }
   496→   496→
   497→   497→  // ---- Step 6: PSU ----
   498→   498→  const psuProducts = allProducts.get('psu') || []
   499→   499→  const psuAlloc = allocations['psu']
   500→   500→
   501→   501→  if (psuAlloc && psuProducts.length > 0) {
   502→   502→    // If we know GPU wattage, prefer PSUs that can handle it
   503→   503→    let psuCompatFilter: ((p: ProductWithArsPrice) => boolean) | undefined
   504→   504→    if (gpuMinWattage) {
   505→   505→      psuCompatFilter = (p: ProductWithArsPrice) => {
   506→   506→        const wattMatch = p.name.match(/(\d{3,4})\s*W/i)
   507→   507→        return wattMatch ? parseInt(wattMatch[1]) >= gpuMinWattage! : true // allow PSUs without wattage in name
   508→   508→      }
   509→   509→    }
   510→   510→
   511→   511→    const psu = pickProductForSlot(psuProducts, psuAlloc * 0.3, psuAlloc * 1.5, psuCompatFilter)
   512→   512→    if (psu) {
   513→   513→      debugLog.push(`PSU: ${psu.name} ($${Math.round(psu.arsComparePrice).toLocaleString()} ARS)`)
   514→   514→      components.push(makeComponent(psu, 'psu', 'Fuente'))
   515→   515→    } else {
   516→   516→      debugLog.push('PSU: none found')
   517→   517→      return null // PSU is required
   518→   518→    }
   519→   519→  } else {
   520→   520→    return null
   521→   521→  }
   522→   522→
   523→   523→  // ---- Step 7: Case (optional - don't fail if not found) ----
   524→   524→  const caseProducts = allProducts.get('case') || []
   525→   525→  const caseAlloc = allocations['case']
   526→   526→
   527→   527→  if (caseAlloc && caseProducts.length > 0) {
   528→   528→    const caseProduct = pickProductForSlot(caseProducts, caseAlloc * 0.2, caseAlloc * 1.5)
   529→   529→    if (caseProduct) {
   530→   530→      debugLog.push(`Case: ${caseProduct.name} ($${Math.round(caseProduct.arsComparePrice).toLocaleString()} ARS)`)
   531→   531→      components.push(makeComponent(caseProduct, 'case', 'Gabinete'))
   532→   532→    } else {
   533→   533→      debugLog.push('Case: none in budget (non-fatal)')
   534→   534→    }
   535→   535→  }
   536→   536→
   537→   537→  // ---- Step 8: Cooling (optional) ----
   538→   538→  if (profile.cooling > 0) {
   539→   539→    const coolingProducts = allProducts.get('cooling') || []
   540→   540→    const coolingAlloc = allocations['cooling']
   541→   541→
   542→   542→    if (coolingAlloc && coolingProducts.length > 0) {
   543→   543→      const cooling = pickProductForSlot(coolingProducts, coolingAlloc * 0.2, coolingAlloc * 1.5)
   544→   544→      if (cooling) {
   545→   545→        debugLog.push(`Cooling: ${cooling.name} ($${Math.round(cooling.arsComparePrice).toLocaleString()} ARS)`)
   546→   546→        components.push(makeComponent(cooling, 'cooling', 'Refrigeración'))
   547→   547→      }
   548→   548→    }
   549→   549→  }
   550→   550→
   551→   551→  // Log the full build
   552→   552→  const totalPrice = components.reduce((sum, c) => sum + c.productComparePrice * c.quantity, 0)
   553→   553→  debugLog.push(`Total: $${Math.round(totalPrice).toLocaleString()} ARS (${components.length} components)`)
   554→   554→  console.log(`[pc-assistant] Build result:\n${debugLog.join('\n')}`)
   555→   555→
   556→   556→  return components
   557→   557→}

   700→   700→export async function POST(request: NextRequest) {
   701→   701→  try {
   702→   702→    // Check feature flag
   703→   703→    const aiEnabled = await isAiEnabled()
   704→   704→    if (!aiEnabled) {
   705→   705→      return NextResponse.json({ error: 'IA deshabilitada' }, { status: 403 })
   706→   706→    }
   707→   707→
   708→   708→    // Parse request
   709→   709→    let body: { messages: ChatMessage[] }
   710→   710→    try {
   711→   711→      body = await request.json()
   712→   712→    } catch {
   713→   713→      return NextResponse.json({ error: 'Body inválido' }, { status: 400 })
   714→   714→    }
   715→   715→
   716→   716→    const { messages } = body
   717→   717→    if (!Array.isArray(messages) || messages.length === 0) {
   718→   718→      return NextResponse.json({ error: 'Se requiere un array de mensajes' }, { status: 400 })
   719→   719→    }
   720→   720→
   721→   721→    // Detect if builds were already shown (post-build conversation mode)
   722→   722→    // If the assistant already returned builds, subsequent messages should be
   723→   723→    // treated as normal chat, not trigger a rebuild
   724→   724→    const assistantMessages = messages.filter(m => m.role === 'assistant')
   725→   725→    const buildsAlreadyShown = assistantMessages.some(m =>
   726→   726→      m.content.includes('opciones') || m.content.includes('Opciones') || m.content.includes('armé') || m.content.includes('Armé')
   727→   727→    )
   728→   728→    const userMessages = messages.filter(m => m.role === 'user')
   729→   729→    const hasMultipleUserMessagesAfterBuild = userMessages.length > 2 // user asked use case + budget + more
   730→   730→
   731→   731→    // Call Groq for chat response
   732→   732→    const controller = new AbortController()
   733→   733→    const timeoutId = setTimeout(() => controller.abort(), 15000)
   734→   734→
   735→   735→    // Use a different system prompt for post-build conversation
   736→   736→    const systemPrompt = buildsAlreadyShown && hasMultipleUserMessagesAfterBuild
   737→   737→      ? POST_BUILD_SYSTEM_PROMPT
   738→   738→      : CHAT_SYSTEM_PROMPT
   739→   739→
   740→   740→    let chatResult: any = null
   741→   741→    try {
   742→   742→      const result = await grokChat({
   743→   743→        messages: [
   744→   744→          { role: 'system', content: systemPrompt },
   745→   745→          ...messages.map(m => ({ role: m.role as 'user' | 'assistant', content: m.content })),
   746→   746→        ],
   747→   747→        temperature: 0.5,
   748→   748→        maxTokens: 400,
   749→   749→        signal: controller.signal,
   750→   750→      })
   751→   751→
   752→   752→      clearTimeout(timeoutId)
   753→   753→
   754→   754→      if (!result.content) {
   755→   755→        return NextResponse.json({
   756→   756→          message: 'Disculpá, no pude procesar tu mensaje. ¿Podés repetirlo?',
   757→   757→          ready: false,
   758→   758→          builds: null,
   759→   759→        })
   760→   760→      }
   761→   761→
   762→   762→      chatResult = parseLlmJson(result.content)
   763→   763→      if (!chatResult) {
   764→   764→        return NextResponse.json({
   765→   765→          message: result.content,
   766→   766→          ready: false,
   767→   767→          builds: null,
   768→   768→        })
   769→   769→      }
   770→   770→    } catch (error) {
   771→   771→      clearTimeout(timeoutId)
   772→   772→      console.error('[pc-assistant] Groq chat error:', error)
   773→   773→      return NextResponse.json({
   774→   774→        message: 'Disculpá, estoy teniendo problemas para conectarme. ¿Podés intentar de nuevo?',
   775→   775→        ready: false,
   776→   776→        builds: null,
   777→   777→      })
   778→   778→    }
   779→   779→
   780→   780→    const responseMessage = chatResult.message || '¿En qué más te puedo ayudar?'
   781→   781→    const isReady = chatResult.ready === true
   782→   782→    const useCase = chatResult.use_case || null
   783→   783→    const budget = typeof chatResult.budget === 'number' ? chatResult.budget : null
   784→   784→
   785→   785→    // If builds were already shown, skip build generation and just chat
   786→   786→    if (buildsAlreadyShown && hasMultipleUserMessagesAfterBuild) {
   787→   787→      return NextResponse.json({
   788→   788→        message: responseMessage,
   789→   789→        ready: false,
   790→   790→        builds: null,
   791→   791→      })
   792→   792→    }
   793→   793→
   794→   794→    // If not ready yet, just return the chat message

   805→   805→
   806→   806→    const allProducts = await fetchAllProductsWithArsPrices()

   822→   822→            components,
   823→   823→          })
