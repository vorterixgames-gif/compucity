   211→   211→/**
   212→   212→ * Build a category lookup from the database.
   213→   213→ * Returns slug -> id map, name -> id map, and parent info for subcategory logic.
   214→   214→ */
   215→   215→async function buildCategoryLookup(): Promise<{
   216→   216→  slugToId: Record<string, string>
   217→   217→  nameToId: Record<string, string>
   218→   218→  idToParentId: Record<string, string | null>
   219→   219→  parentSlugToChildSlugs: Record<string, string[]>
   220→   220→}> {
   221→   221→  const result = await db.execute('SELECT id, name, slug, parentId FROM categories')
   222→   222→  const slugToId: Record<string, string> = {}
   223→   223→  const nameToId: Record<string, string> = {}
   224→   224→  const idToParentId: Record<string, string | null> = {}
   225→   225→  const parentSlugToChildSlugs: Record<string, string[]> = {}
   226→   226→  for (const row of result.rows as any[]) {
   227→   227→    if (row.slug) slugToId[row.slug] = row.id
   228→   228→    if (row.name) nameToId[row.name.toLowerCase()] = row.id
   229→   229→    idToParentId[row.id] = row.parentId || null
   230→   230→    if (row.parentId) {
   231→   231→      // Find parent slug
   232→   232→      const parentRow = (result.rows as any[]).find((r: any) => r.id === row.parentId)
   233→   233→      if (parentRow?.slug) {
   234→   234→        if (!parentSlugToChildSlugs[parentRow.slug]) parentSlugToChildSlugs[parentRow.slug] = []
   235→   235→        parentSlugToChildSlugs[parentRow.slug].push(row.slug)
   236→   236→      }
   237→   237→    }
   238→   238→  }
   239→   239→  return { slugToId, nameToId, idToParentId, parentSlugToChildSlugs }
   240→   240→}

   246→   246→async function buildSupplierMappingLookup(supplierId: string): Promise<Record<string, string>> {
   247→   247→  const result = await db.execute({
   248→   248→    sql: 'SELECT supplierCategory, storeCategoryId FROM supplier_category_mappings WHERE supplierId = ?',
   249→   249→    args: [supplierId],
   250→   250→  })
   251→   251→  const map: Record<string, string> = {}
   252→   252→  for (const row of result.rows as any[]) {
   253→   253→    map[row.supplierCategory] = row.storeCategoryId
   254→   254→  }
   255→   255→  return map
   256→   256→}

   258→   258→/**
   259→   259→ * Map a product to a store category using:
   260→   260→ * 1. Supplier category mapping (if available)
   261→   261→ * 2. Keyword matching with slug-based lookup
   262→   262→ * 3. Subcategory refinement: if the matched category is a parent with subcategories,
   263→   263→ *    use subcategory rules to find the best subcategory match
   264→   264→ * 4. Default: null (no category)
   265→   265→ */
   266→   266→function mapProductToCategory(
   267→   267→  productName: string,
   268→   268→  supplierCategory: string | null,
   269→   269→  supplierMappings: Record<string, string>,
   270→   270→  slugToId: Record<string, string>,
   271→   271→  idToParentId: Record<string, string | null>,
   272→   272→  parentSlugToChildSlugs: Record<string, string[]>,
   273→   273→  slugToParentSlug?: Record<string, string>
   274→   274→): { categoryId: string | null; method: string } {
   275→   275→  let matchedCategoryId: string | null = null
   276→   276→  let method = 'none'
   277→   277→
   278→   278→  // 1. Check supplier category mapping first
   279→   279→  if (supplierCategory && supplierMappings[supplierCategory]) {
   280→   280→    matchedCategoryId = supplierMappings[supplierCategory]
   281→   281→    method = 'mapping'
   282→   282→  }
   283→   283→
   284→   284→  // 2. Keyword matching (only if no mapping match)
   285→   285→  if (!matchedCategoryId) {
   286→   286→    const name = (productName || '').toUpperCase()
   287→   287→    for (const mapping of CATEGORY_KEYWORD_MAP) {
   288→   288→      if (mapping.keywords.some(kw => name.includes(kw))) {
   289→   289→        const categoryId = slugToId[mapping.categorySlug]
   290→   290→        if (categoryId) {
   291→   291→          matchedCategoryId = categoryId
   292→   292→          method = 'keyword'
   293→   293→          break
   294→   294→        }
   295→   295→      }
   296→   296→    }
   297→   297→  }
   298→   298→
   299→   299→  if (!matchedCategoryId) {
   300→   300→    return { categoryId: null, method: 'none' }
   301→   301→  }
   302→   302→
   303→   303→  // 2.5. Category correction: if a product was mapped to a category by supplier mapping
   304→   304→  // but its name indicates it should be in a different category, correct it.
   305→   305→  // This handles cases like "Plataforma AMD/Intel" mapping to microprocesadores
   306→   306→  // but containing both processors and motherboards.
   307→   307→  const upperName = (productName || '').toUpperCase()
   308→   308→  const CATEGORY_CORRECTIONS: { nameKeyword: string; targetSlug: string; sourceSlugs: string[] }[] = [
   309→   309→    {
   310→   310→      // Motherboards incorrectly mapped to microprocesadores
   311→   311→      nameKeyword: 'MOTHER',
   312→   312→      targetSlug: 'motherboards',
   313→   313→      sourceSlugs: ['microprocesadores'],

  1208→  1208→/**
  1209→  1209→ * Strip PHP notices/warnings/errors from Air Intra API response text.
  1210→  1210→ * Air Intra's PHP server injects HTML-formatted PHP notices like:
  1211→  1211→ *   <br /> <b>Notice</b>: Undefined property: stdClass::$estado in /path/file.php on line 54
  1212→  1212→ * These can appear BEFORE, AFTER, or INSIDE the JSON array (between product objects).
  1213→  1213→ * This function removes them while preserving the actual JSON data.
  1214→  1214→ */
  1215→  1215→function stripPhpNotices(text: string): string {
  1216→  1216→  let cleaned = text
  1217→  1217→    // Remove complete PHP notice/warning/error blocks (HTML format with <b> tags)
  1218→  1218→    // Pattern: optional <br />, then <b>Type</b>: message in /path/file.php on line NNN
  1219→  1219→    .replace(/(?:<br\s*\/?>\s*)?<b>(?:Notice|Warning|Fatal error|Parse error|Deprecated)<\/b>:\s*.*?on line \d+\s*/gis, '')
  1220→  1220→    // Remove plain-text PHP notices (without HTML tags)
  1221→  1221→    .replace(/(?:^|\n)\s*(?:Notice|Warning|Fatal error|Parse error|Deprecated):\s*.*?on line \d+\s*/gis, '')
  1222→  1222→    // Remove any remaining standalone <br /> tags
  1223→  1223→    .replace(/<br\s*\/?>\s*/gi, '')
  1224→  1224→    // Remove leftover <b> or </b> tags
  1225→  1225→    .replace(/<\/?b>/gi, '')
  1226→  1226→    // Fix trailing commas before ] or } (can happen after removing notices)
  1227→  1227→    .replace(/,\s*([}\]])/g, '$1')
  1228→  1228→
  1229→  1229→  return cleaned.trim()
  1230→  1230→}

  1333→  1333→  while (i < text.length) {

  1335→  1335→    if (text[i] !== '{') {
  1336→  1336→      i++

  1338→  1338→    }

  1351→  1351→        continue

  1354→  1354→        esc = true
  1355→  1355→        continue
  1356→  1356→      }

  1362→  1362→

  1364→  1364→      else if (ch === '}') {
  1365→  1365→        depth--
  1366→  1366→        if (depth === 0) {
  1367→  1367→          objEnd = j

  1370→  1370→      }

  1373→  1373→    if (objEnd === -1) {
  1374→  1374→      i++

  1377→  1377→
  1378→  1378→    const objText = text.substring(i, objEnd + 1)

  1380→  1380→    // Quick check: skip objects that are clearly not products
  1381→  1381→    // (e.g., error objects, impuesto_iva fragments, etc.)
  1382→  1382→    if (objText.includes('"codigo"') || objText.includes('"codiart"')) {
  1383→  1383→      try {
  1384→  1384→        const obj = JSON.parse(objText)
  1385→  1385→        products.push(obj)
  1386→  1386→      } catch {
  1387→  1387→        // Object itself is corrupted, skip it
  1388→  1388→      }
  1389→  1389→    }
  1390→  1390→
  1391→  1391→    i = objEnd + 1
  1392→  1392→  }
  1393→  1393→
  1394→  1394→  return products
  1395→  1395→}
